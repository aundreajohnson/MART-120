// ------------------------------ // Simple Game Project // Beginner Version // ------------------------------ // player position var playerX = 100; var playerY = 100; // movement keys var w = 87; var s = 83; var a = 65; var d = 68; // obstacles var obstacle1X = 50; var obstacle1Y = 100; var obstacle2X = 300; var obstacle2Y = 200; var obstacle1XSpeed; var obstacle1YSpeed; var obstacle2XSpeed; var obstacle2YSpeed; // mouse shape var mouseShapeX = -50; var mouseShapeY = -50; function setup() { createCanvas(500, 500); // random speeds for obstacles obstacle1XSpeed = random(1, 5); obstacle1YSpeed = random(1, 5); obstacle2XSpeed = random(1, 5); obstacle2YSpeed = random(1, 5); } function draw() { background(120, 180, 200); // function calls only createBorders(10); createExit(); drawPlayer(); movePlayer(); drawObstacles(); moveObstacles(); drawMouseShape(); displayWinMessage(); } // create the player function drawPlayer() { fill(0, 0, 255); circle(playerX, playerY, 25); } // move the player function movePlayer() { if (keyIsDown(w)) playerY -= 5; if (keyIsDown(s)) playerY += 5; if (keyIsDown(a)) playerX -= 5; if (keyIsDown(d)) playerX += 5; } // draw the obstacles function drawObstacles() { fill(255, 0, 0); rect(obstacle1X, obstacle1Y, 40, 40); fill(0, 255, 0); ellipse(obstacle2X, obstacle2Y, 50, 50); } // move the obstacles function moveObstacles() { obstacle1X += obstacle1XSpeed; obstacle1Y += obstacle1YSpeed; obstacle2X += obstacle2XSpeed; obstacle2Y += obstacle2YSpeed; // wrap around edges if (obstacle1X > width) obstacle1X = 0; if (obstacle1Y > height) obstacle1Y = 0; if (obstacle2X > width) obstacle2X = 0; if (obstacle2Y > height) obstacle2Y = 0; } // draw shape when mouse is clicked function drawMouseShape() { fill(100, 100, 120); circle(mouseShapeX, mouseShapeY, 30); } // make the borders function createBorders(thickness) { fill(0); noStroke(); rect(0, 0, width, thickness); // top rect(0, 0, thickness, height); // left rect(0, height - thickness, width, thickness); // bottom rect(width - thickness, 0, thickness, height - 50); // right (leave opening) } // make the exit function createExit() { textSize(16); fill(0); text("EXIT", width - 60, height - 30); } // show "You Win" message function displayWinMessage() { if (playerX > width - 20 && playerY > height - 50) { fill(0); textSize(26); text("You Win!", width / 2 - 60, height / 2); } } // create shape when mouse is clicked function mouseClicked() { mouseShapeX = mouseX; mouseShapeY = mouseY; }
