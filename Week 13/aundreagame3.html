// ------------------------------ // Simple Game with Arrays // ------------------------------ // player position var playerX = 100; var playerY = 100; // movement keys var w = 87; var s = 83; var a = 65; var d = 68; // arrays for obstacles var obstacleX = []; var obstacleY = []; var obstacleSize = []; var obstacleType = []; // 0 = square, 1 = circle var obstacleXSpeed = []; var obstacleYSpeed = []; var obstacleColor = []; // mouse-added shape var mouseShapeX = -50; var mouseShapeY = -50; function setup() { createCanvas(500, 500); // create 5 obstacles for (var i = 0; i < 5; i++) { obstacleX[i] = random(50, 450); obstacleY[i] = random(50, 450); obstacleSize[i] = random(30, 50); obstacleType[i] = floor(random(0, 2)); // square or circle obstacleXSpeed[i] = random(1, 4); obstacleYSpeed[i] = random(1, 4); obstacleColor[i] = color(random(255), random(255), random(255)); } } function draw() { background(120, 180, 200); createBorders(10); createExit(); drawPlayer(); movePlayer(); drawObstacles(); moveObstacles(); drawMouseShape(); displayWinMessage(); } // draw the player function drawPlayer() { fill(0, 0, 255); circle(playerX, playerY, 25); } // move the player function movePlayer() { if (keyIsDown(w)) playerY -= 5; if (keyIsDown(s)) playerY += 5; if (keyIsDown(a)) playerX -= 5; if (keyIsDown(d)) playerX += 5; } // draw all obstacles function drawObstacles() { for (var i = 0; i < obstacleX.length; i++) { fill(obstacleColor[i]); if (obstacleType[i] === 0) { // square rect(obstacleX[i], obstacleY[i], obstacleSize[i], obstacleSize[i]); } else { // circle ellipse(obstacleX[i], obstacleY[i], obstacleSize[i], obstacleSize[i]); } } } // move all obstacles function moveObstacles() { for (var i = 0; i < obstacleX.length; i++) { obstacleX[i] += obstacleXSpeed[i]; obstacleY[i] += obstacleYSpeed[i]; // wrap around screen if (obstacleX[i] > width) obstacleX[i] = 0; if (obstacleX[i] < 0) obstacleX[i] = width; if (obstacleY[i] > height) obstacleY[i] = 0; if (obstacleY[i] < 0) obstacleY[i] = height; } } // draw mouse-clicked obstacle function drawMouseShape() { fill(100, 100, 120); circle(mouseShapeX, mouseShapeY, 30); } // borders function createBorders(thickness) { fill(0); noStroke(); rect(0, 0, width, thickness); // top rect(0, 0, thickness, height); // left rect(0, height - thickness, width, thickness); // bottom rect(width - thickness, 0, thickness, height - 50); // right (gap for exit) } // exit label function createExit() { textSize(16); fill(0); text("EXIT", width - 60, height - 30); } // win message function displayWinMessage() { if (playerX > width - 20 && playerY > height - 50) { fill(0); textSize(26); text("You Win!", width / 2 - 60, height / 2); } } // add new obstacle on click function mouseClicked() { mouseShapeX = mouseX; mouseShapeY = mouseY; // also add to arrays obstacleX.push(mouseX); obstacleY.push(mouseY); obstacleSize.push(30); obstacleType.push(floor(random(0, 2))); obstacleXSpeed.push(random(1, 4)); obstacleYSpeed.push(random(1, 4)); obstacleColor.push(color(random(255), random(255), random(255))); }
