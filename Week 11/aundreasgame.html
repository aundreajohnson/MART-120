// ------------------------------- // Simple Game Project // ------------------------------- // player position var playerX = 100; var playerY = 100; // player movement keys var w = 87; var s = 83; var a = 65; var d = 68; // obstacles var obstacle1X = 50; var obstacle1Y = 100; var obstacle2X = 300; var obstacle2Y = 200; var obstacle1XSpeed; var obstacle1YSpeed; var obstacle2XSpeed; var obstacle2YSpeed; // mouse obstacle var mouseObstacleX; var mouseObstacleY; // setup canvas and starting values function setup() { createCanvas(500, 500); // random speeds for obstacles obstacle1XSpeed = random(1, 5); obstacle1YSpeed = random(1, 5); obstacle2XSpeed = random(1, 5); obstacle2YSpeed = random(1, 5); } // draw loop function draw() { background(120, 180, 200); // call border and exit functions createBorders(10); drawExit(); // draw and move player drawPlayer(); movePlayer(); // draw and move obstacles drawObstacles(); moveObstacles(); // draw mouse obstacle (if added) fill(100, 100, 120); circle(mouseObstacleX, mouseObstacleY, 30); // check if player reaches the exit if (playerX > width - 20 && playerY > height - 50) { fill(0); textSize(26); text("You Win!", width / 2 - 60, height / 2); } } // draw player function drawPlayer() { fill(0, 0, 255); circle(playerX, playerY, 25); } // player movement function movePlayer() { if (keyIsDown(w)) { playerY -= 5; } if (keyIsDown(s)) { playerY += 5; } if (keyIsDown(a)) { playerX -= 5; } if (keyIsDown(d)) { playerX += 5; } } // draw obstacles function drawObstacles() { fill(255, 0, 0); rect(obstacle1X, obstacle1Y, 40, 40); fill(0, 255, 0); ellipse(obstacle2X, obstacle2Y, 50, 50); } // move obstacles randomly function moveObstacles() { obstacle1X += obstacle1XSpeed; obstacle1Y += obstacle1YSpeed; obstacle2X += obstacle2XSpeed; obstacle2Y += obstacle2YSpeed; // wrap around screen edges if (obstacle1X > width) obstacle1X = 0; if (obstacle1X < 0) obstacle1X = width; if (obstacle1Y > height) obstacle1Y = 0; if (obstacle1Y < 0) obstacle1Y = height; if (obstacle2X > width) obstacle2X = 0; if (obstacle2X < 0) obstacle2X = width; if (obstacle2Y > height) obstacle2Y = 0; if (obstacle2Y < 0) obstacle2Y = height; } // borders and exit area function createBorders(thickness) { // top rect(0, 0, width, thickness); // left rect(0, 0, thickness, height); // bottom rect(0, height - thickness, width, thickness); // right (leaves opening for exit) rect(width - thickness, 0, thickness, height - 50); } // draw exit area function drawExit() { textSize(16); fill(0); text("EXIT", width - 60, height - 30); } // add obstacle when mouse is clicked function mouseClicked() { mouseObstacleX = mouseX; mouseObstacleY = mouseY; }
